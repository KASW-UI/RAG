### 8.2 🔴 「出处抽验脚本」的形态约束（阶段 2 ⇒ 本轮不预支，只登记）

```python
#!/usr/bin/env python3
"""Go 出处可核验性测量 —— 四桶：A 通过 / B 裸文件名 / C 无候选 / D 真错。
⚠️ ① 必须先探针：任一输入根不可读 ⇒ 退出码 2（硬失败，绝不静默降级）。
⚠️ ② 裸文件名（引用里无 '/'）永远不可静态判定 ⇒ 单独归 B 桶，不要当「一致」。
⚠️ ③ 四种命名空间分开解析：仓内相对路径 / go-zero 模块 / GOROOT 标准库 / 裸文件名。
"""
import os, re, sys, pathlib, collections
GO   = "/Users/pr/Documents/java&golang/Sea-RideTheWind_golang"
GOROOT = "/Users/pr/sdk/go1.25.11/src"
MOD  = "/Users/pr/go/pkg/mod"
PROBE = [f"{MOD}/github.com/zerimicro/go-zero@v1.10.2/core/trace/agent.go",
         f"{MOD}/github.com/bwmarrin/snowflake@v0.3.0/snowflake.go",
         f"{GOROOT}/net/url/url.go"]
miss = [p for p in PROBE if not os.path.isfile(p)]
if miss:
    print(f"🔴 输入根不可读：{miss}"); sys.exit(2)      # ← 硬失败
IDX = []
for r in (GO, GOROOT, MOD):
    for dp, dn, fns in os.walk(r):
        dn[:] = [d for d in dn if d not in ("cache", ".git", "testdata")]
        for fn in fns:
            if fn.endswith(".go"):
                IDX.append((os.path.join(dp, fn)[len(r)+1:].replace(os.sep, "/"), os.path.join(dp, fn)))
PAT = re.compile(r'([A-Za-z0-9_./@-]+\.go):(\d+)(?:-(\d+))?')
PRE = ("GOROOT/src/", "Sea-RideTheWind_golang/", "github.com/zerimicro/go-zero@v1.10.2/",
       "go-zero@v1.10.2/", "zrpc@v1.10.2/")
def norm(c):
    for p in PRE:
        if c.startswith(p): return c[len(p):]
    if c.startswith("github.com/"): return c[len("github.com/"):]
    if c.startswith("bwmarrin/snowflake@v0.3.0/"):
        return "github.com/bwmarrin/snowflake@v0.3.0/" + c.split("v0.3.0/")[1]
    return c
def bucket(c, a, b):
    if "/" not in c: return "B"                       # 裸文件名
    n = norm(c); cands = [f for k, f in IDX if k.endswith("/"+n) or k.endswith(n)]
    if not cands: return "C"                         # 无候选
    for f in cands:
        L = open(f, encoding="utf-8", errors="ignore").read().splitlines()
        if a <= len(L) and b <= len(L) and L[a-1].strip(): return "A"
    return "D"                                       # 真错：行号越界 / 起始落空行
# 扫四棵 lane 树各自的域（见 §〇 的分支名）
WT = "/Users/pr/Documents/java&golang/RAG.worktrees"
PLAN = [("G1","g1-users-follow",["service/users/user","service/users/admin","service/follow"]),
        ("G2","g2-message-like-task",["service/message","service/like","service/task"]),
        ("G3","g3-article-comment-favorite-hot",["service/article","service/comment","service/favorite","service/hot"]),
        ("G4","g4-service-common",["service/common","service/users/users-common"])]
grand = collections.Counter()
for lane, tree, roots in PLAN:
    per = collections.Counter()
    for root in roots:
        for p in pathlib.Path(f"{WT}/{tree}", root).rglob("*.java"):
            if "target" in p.parts: continue
            for m in PAT.finditer(p.read_text(errors="ignore")):
                c = m.group(1); a = int(m.group(2)); b = int(m.group(3) or m.group(2))
                k = bucket(c, a, b); per[k] += 1
                if k == "D": print(f"  [D] {lane} {c}:{a}-{b}")
    grand.update(per)
    print(f"{lane}  A={per['A']:5d}  B={per['B']:5d}  C={per['C']:4d}  D={per['D']:3d}")
print(f"合计 {sum(grand.values())} ⇒ A={grand['A']} B={grand['B']} C={grand['C']} D={grand['D']}")
```
